### 8.1 页面可访问性测试脚本

```bash
#!/bin/bash
# 测试所有页面 HTTP 状态码

# 登录获取 Token
API_LOGIN=$(curl -s -X POST http://localhost:3001/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"admin@example.com","password":"<ADMIN_PASSWORD>"}')
TOKEN=$(echo "$API_LOGIN" | grep -oP '"token":"[^"]*"' | head -1 | cut -d'"' -f4)

# 页面列表
PAGES=(
  "admin" "admin/users" "admin/groups" "admin/models" "admin/keypool" 
  "admin/providers" "admin/subscriptions" "admin/redemption" "admin/audit-logs" 
  "admin/settings" "admin/enterprise" "admin/monitor-upstream" "admin/monitor-downstream"
  "super/admin-management" "super/providers" "super/payment"
  "enterprise" "enterprise/users" "enterprise/keys" "enterprise/billing"
  "dashboard" "apikeys" "billing" "orders" "invoices" "profile"
)

PASS=0
FAIL=0
for page in "${PAGES[@]}"; do
  RESP=$(curl -s -b "jwt=$TOKEN" -o /dev/null -w "%{http_code}" "http://localhost:3001/$page")
  if [ "$RESP" == "200" ]; then
    echo "✓ $page"
    ((PASS++))
  else
    echo "✗ $page -> $RESP"
    ((FAIL++))
  fi
done

echo ""
echo "=== 结果: $PASS 通过, $FAIL 失败 ==="
```

### 8.5 数据库完整性检查脚本

```bash
#!/bin/bash
# 检查数据库表和字段完整性

echo "=== 数据库完整性检查 ==="

echo "--- 1. 核心表检查 ---"
TABLES=("users" "api_keys" "user_groups" "enterprises" "enterprise_members" "upstream_keys" "service_models" "providers" "subscription_plans" "redemption_codes" "audit_logs" "system_settings" "orders" "invoices")

for table in "${TABLES[@]}"; do
  EXISTS=$(docker exec ai-portal-postgres psql -U postgres -d ai_proxy_portal -t -c "SELECT EXISTS (SELECT FROM information_schema.tables WHERE table_name = '$table');" | tr -d ' ')
  if [ "$EXISTS" == "t" ]; then
    echo "✓ $table"
  else
    echo "✗ $table (缺失)"
  fi
done

echo ""
echo "--- 2. user_groups 字段检查 ---"
docker exec ai-portal-postgres psql -U postgres -d ai_proxy_portal -c "\d user_groups" | grep -E "tpm|tph|tpd|tpw|tpmonth|rph|rp5h|rpw|rpmonth" | while read line; do
  FIELD=$(echo "$line" | awk '{print $1}')
  TYPE=$(echo "$line" | awk '{print $2}')
  if [[ "$TYPE" == *"bigint"* ]]; then
    echo "✓ $FIELD ($TYPE)"
  else
    echo "✗ $FIELD ($TYPE) - 应为 bigint"
  fi
done

echo ""
echo "--- 3. api_keys key_type 约束检查 ---"
docker exec ai-portal-postgres psql -U postgres -d ai_proxy_portal -c "\d api_keys" | grep -A1 "key_type_check" | head -2

echo ""
echo "--- 4. 数据统计 ---"
docker exec ai-portal-postgres psql -U postgres -d ai_proxy_portal -c "
SELECT 'users' as table_name, COUNT(*) as count FROM users
UNION ALL SELECT 'api_keys', COUNT(*) FROM api_keys
UNION ALL SELECT 'user_groups', COUNT(*) FROM user_groups
UNION ALL SELECT 'enterprises', COUNT(*) FROM enterprises
UNION ALL SELECT 'providers', COUNT(*) FROM providers
UNION ALL SELECT 'service_models', COUNT(*) FROM service_models
UNION ALL SELECT 'upstream_keys', COUNT(*) FROM upstream_keys;
"

echo ""
echo "=== 检查完成 ==="
```
