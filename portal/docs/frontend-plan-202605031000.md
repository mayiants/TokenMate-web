# AI Portal Go - Frontend Plan (Standalone)

> **ACTIVE** - Created: 2026-05-03

---

## 目标

`ai-portal-go` 是完全独立的项目，前端从零开始，不依赖 `ai-proxy-main`。

---

## 技术栈

| 层 | 技术 | 理由 |
|----|------|------|
| 框架 | React 18 + TypeScript | 现代、生态好 |
| 构建 | Vite | 快速 HMR |
| 样式 | Tailwind CSS v4 | 实用优先 |
| 路由 | React Router v6 | 标准 |
| HTTP | fetch (原生) | 无需 Axios |
| 图标 | Lucide React | 轻量 |
| 图表 | Recharts | React 生态 |
| 状态 | Zustand | 轻量状态管理 |
| UI 组件 | shadcn/ui | 可定制 |

## API 客户端

```typescript
// src/lib/api.ts
const BASE = '/api';

async function request<T>(path: string, opts?: RequestInit): Promise<T> {
  const token = localStorage.getItem('token');
  const res = await fetch(`${BASE}${path}`, {
    ...opts,
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...opts?.headers,
    },
  });
  if (!res.ok) throw await res.json();
  return res.json();
}

export const api = {
  // Auth
  login: (email: string, password: string) =>
    request('/auth/login', { method: 'POST', body: JSON.stringify({ email, password }) }),
  register: (data: { name: string; email: string; password: string; code: string }) =>
    request('/auth/register', { method: 'POST', body: JSON.stringify(data) }),
  me: () => request('/auth/me'),

  // API Keys
  listKeys: () => request('/apikeys'),
  createKey: (data: { name: string }) =>
    request('/apikeys', { method: 'POST', body: JSON.stringify(data) }),
  deleteKey: (id: string) =>
    request(`/apikeys/${id}`, { method: 'DELETE' }),

  // Billing
  balance: () => request('/billing/balance'),
  usage: () => request('/billing/usage-details'),

  // Admin
  adminUsers: () => request('/admin/users'),
  adminProviders: () => request('/admin/providers'),
  adminModels: () => request('/admin/service-models'),
  adminSettings: () => request('/admin/settings'),
};
```

## Go 后端集成

### 嵌入前端

```go
//go:embed dist/*
var frontendFS embed.FS

func main() {
    router := gin.Default()

    // API routes
    api := router.Group("/api")
    // ... register routes ...

    // Frontend static files
    distFS, _ := fs.Sub(frontendFS, "dist")
    router.StaticFS("/assets", http.FS(distFS))
    router.NoRoute(func(c *gin.Context) {
        if !strings.HasPrefix(c.Request.URL.Path, "/api") {
            c.FileFromFS("/index.html", http.FS(distFS))
        }
    })
}
```

### Dockerfile

```dockerfile
# Stage 1: Build frontend
FROM node:20-alpine AS frontend
WORKDIR /app
COPY web/package*.json ./
RUN npm ci
COPY web/ ./
RUN npm run build

# Stage 2: Build Go backend
FROM golang:1.21-alpine AS builder
WORKDIR /app
COPY go.mod go.sum ./
RUN go mod download
COPY . .
COPY --from=frontend /app/dist ./dist/
RUN CGO_ENABLED=0 go build -o portal ./cmd/server

# Stage 3: Final
FROM alpine:3.19
WORKDIR /app
COPY --from=builder /app/portal .
COPY --from=builder /app/migrations ./migrations/
EXPOSE 3001
ENTRYPOINT ["./portal"]
```
